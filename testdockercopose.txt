services:
  postgres:
    image: postgres:17.3
    container_name: postgres_container
    restart: always
    environment:
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}
    ports:
      - "5432:5432"
    volumes:
      - ./postgres_data:/var/lib/postgresql/data
    networks:
      - app-network

  pgadmin:
    image: dpage/pgadmin4:2025-07-21-1
    container_name: pgadmin_container
    restart: always
    depends_on:
      - postgres
    ports:
      - "5050:80"
    environment:
      PGADMIN_DEFAULT_EMAIL: ${PGADMIN_DEFAULT_EMAIL}
      PGADMIN_DEFAULT_PASSWORD: ${PGADMIN_DEFAULT_PASSWORD}
    networks:
      - app-network
    volumes:
      - ./pgadmin:/var/lib/pgadmin

  springboot-dev:
    build: .
    container_name: springboot_dev
    volumes:
      - ./src:/app/src
      - ./pom.xml:/app/pom.xml
      - ~/.m2:/root/.m2
      - ./uploads:/app/uploads         # Carpeta para archivos subidos
      - ./logs:/app/logs 
    working_dir: /app
    ports:
      - "8080:8080"
    command: mvn spring-boot:run
    environment:
      - DB_HOST=postgres
      - DB_PORT=5432
      - DB_NAME=${POSTGRES_DB}
      - DB_USER=${POSTGRES_USER}
      - DB_PASS=${POSTGRES_PASSWORD}
      - SPRING_PROFILES_ACTIVE=dev
    restart: unless-stopped
    networks:
      - app-network
  springboot-prod:
    build:
      context: .
      dockerfile: Dockerfile.prod
    container_name: springboot_prod
    volumes:
      - ./uploads:/app/uploads         # Carpeta para archivos subidos
      - ./logs:/app/logs 
    ports:
      - "8081:8080"
    environment:
      - SPRING_PROFILES_ACTIVE=prod
      - DB_HOST=postgres
      - DB_PORT=5432
      - DB_NAME=${POSTGRES_DB}
      - DB_USER=${POSTGRES_USER}
      - DB_PASS=${POSTGRES_PASSWORD}
    depends_on:
      - postgres
    restart: unless-stopped
    networks:
      - app-network

networks:
  app-network:
    driver: bridge
